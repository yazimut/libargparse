Custom:
    '!'      Индикатор
    '$'      Разделитель значения
    '!!'     Маркер конца опций
'!'         Позиционный аргумент
'!!'        Маркер
'!.+($.*)?' Опция с возможным значением
'.+($.*)?'  Позиционный аргумент


Unix:
"-"             просто позиционный аргумент
"--"            маркер конца опций
                Дальше только позиционные
"-.=?.*"        Короткая опция {.}, после которой,
                возможно через {=}, идёт значение
"--.+(=.*)?"    Длинная опция {.}, после которой
                возможно идёт значение через {=}

Windows:
"/"             Просто позиционный аргумент
"/.+(:.*)?"     Опция {.}, после которой
                возможно идёт значение через {:}
